CI/CD Workflow (ci-cd.yml)

This workflow runs tests, builds the React app, uploads the build artifact, and on pushes to main builds & pushes a Docker image and optionally deploys.
name: CI / CD Pipeline

on:
  push:
    branches: ["**"]
  pull_request:
    branches: ["**"]
  workflow_dispatch:

concurrency:
  group: ${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read
  packages: write
  pages: write
  id-token: write

env:
  NODE_VERSION: 18
  REGISTRY: docker.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  ci:
    name: CI — lint / test / build
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Use Node.js ${{ env.NODE_VERSION }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run linter
        run: |
          if npm run -s lint ; then
            echo "lint ok"
          else
            echo "lint failed"
            exit 1
          fi

      - name: Run tests
        run: |
          if npm test -- --watchAll=false ; then
            echo "tests passed"
          else
            echo "tests failed"
            exit 1
          fi

      - name: Build (production)
        run: npm run build

      - name: Upload built artifact
        uses: actions/upload-artifact@v4
        with:
          name: build
          path: build

  docker:
    name: Docker Build & Push
    runs-on: ubuntu-latest
    needs: ci
    if: github.event_name == 'push' && (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master')

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Log in to Docker Hub
        uses: docker/login-action@v2
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ secrets.DOCKERHUB_USERNAME }}
          password: ${{ secrets.DOCKERHUB_TOKEN }}

      - name: Set up QEMU
        uses: docker/setup-qemu-action@v2

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v2

      - name: Build and push Docker image
        uses: docker/build-push-action@v4
        with:
          context: .
          push: true
          platforms: linux/amd64,linux/arm64
          tags: |
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:latest
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}
          file: ./Dockerfile

  deploy_pages:
    name: Deploy → GitHub Pages
    runs-on: ubuntu-latest
    needs: ci
    if: github.event_name == 'push' && (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master')

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifact
        uses: actions/download-artifact@v4
        with:
          name: build
          path: ./build

      - name: Deploy to GitHub Pages
        uses: peaceiris/actions-gh-pages@v4
        with:
          github_token: ${{ secrets.GITHUB_PAGES_PAT || github.token }}
          publish_dir: ./build
          publish_branch: gh-pages
          user_name: github-actions[bot]
          user_email: 41898282+github-actions[bot]@users.noreply.github.com

  deploy_ssh:
    name: Deploy → SSH (optional)
    runs-on: ubuntu-latest
    needs: ci
    if: github.event_name == 'push' && (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master') && secrets.SSH_PRIVATE_KEY

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Download build artifact
        uses: actions/download-artifact@v4
        with:
          name: build
          path: ./build

      - name: Copy build to remote server
        uses: appleboy/ssh-action@v0.1.8
        with:
          host: ${{ secrets.SSH_HOST }}
          username: ${{ secrets.SSH_USERNAME }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          port: 22
          source: "./build/"
          target: ${{ secrets.REMOTE_DEPLOY_PATH }}
          rm: true

      - name: Remote restart (example)
        uses: appleboy/ssh-action@v0.1.8
        with:
          host: ${{ secrets.SSH_HOST }}
          username: ${{ secrets.SSH_USERNAME }}
          key: ${{ secrets.SSH_PRIVATE_KEY }}
          script: |
            sudo systemctl restart nginx || true
            # or docker-compose -f /path/to/docker-compose.yml pull --quiet && docker-compose -f /path/to/docker-compose.yml up -d --remove-orphans

How to use

Quick steps to enable the pipeline in your repo.
1) Save the YAML to: .github/workflows/ci-cd.yml
2) Add required secrets in GitHub repo Settings -> Secrets:
   - DOCKERHUB_USERNAME
   - DOCKERHUB_TOKEN
   - (optional) GITHUB_PAGES_PAT
   - (optional) SSH_HOST, SSH_USERNAME, SSH_PRIVATE_KEY, REMOTE_DEPLOY_PATH
3) Ensure package.json contains scripts:
   - \"lint\" (optional), \"test\", \"build\"
4) Push to GitHub. CI runs on push & PR; Docker build/push triggers for pushes to main.

Tip: adapt the workflow to your needs — replace Docker Hub steps with ECR/GCR/ACR actions, or remove deploy jobs if not needed.
